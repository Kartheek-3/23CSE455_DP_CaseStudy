% Language Paradigm Insights in 4 Columns
\begin{multicols}{4}

\begin{tcolorbox}[title=\textbf{Java (OOP / JVM)},colback=white,colframe=headerblue!80,arc=2mm,boxrule=0.5pt,top=1mm,bottom=1mm,left=1.5mm,right=1.5mm]
\scriptsize
\begin{itemize}[leftmargin=*,noitemsep]
\item \textbf{Type Safety}: Strict interfaces, generics, explicit type checks.
\item \textbf{Encapsulation}: Clean private/nested classes.
\item \textbf{LOC}: Highest (408 total) due to boilerplate contracts.
\item \textbf{Latency}: 2.5--22.4 ms due to JVM classloader, reflection overhead, and JIT warmup.
\end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[title=\textbf{Python (Dynamic / ABC)},colback=white,colframe=headerblue!80,arc=2mm,boxrule=0.5pt,top=1mm,bottom=1mm,left=1.5mm,right=1.5mm]
\scriptsize
\begin{itemize}[leftmargin=*,noitemsep]
\item \textbf{Expressiveness}: ABC classes, first-class closures, native dicts.
\item \textbf{Duck Typing}: High velocity and minimal boilerplate.
\item \textbf{LOC}: Low footprint (358 total, -12\% vs Java).
\item \textbf{Latency}: 0.01--0.05 ms in micro-execution; low interpreter initialization cost.
\end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[title=\textbf{JavaScript (ESM / V8)},colback=white,colframe=headerblue!80,arc=2mm,boxrule=0.5pt,top=1mm,bottom=1mm,left=1.5mm,right=1.5mm]
\scriptsize
\begin{itemize}[leftmargin=*,noitemsep]
\item \textbf{Native Match}: Closest to source SvelteKit/Node.js architecture.
\item \textbf{Prototypes}: Class syntax with dynamic polymorphism.
\item \textbf{LOC}: Compact (368 total).
\item \textbf{Latency}: 0.14--0.57 ms; V8 JIT inline caching provides steady predictable performance.
\end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[title=\textbf{C++ (Native / RAII)},colback=white,colframe=headerblue!80,arc=2mm,boxrule=0.5pt,top=1mm,bottom=1mm,left=1.5mm,right=1.5mm]
\scriptsize
\begin{itemize}[leftmargin=*,noitemsep]
\item \textbf{Zero-Overhead}: Smart pointers (\texttt{shared\_ptr}), deterministic memory.
\item \textbf{Vtable}: Compile-time optimization with \texttt{-O2}.
\item \textbf{LOC}: Lowest production footprint (245 LOC, -40\% vs Java).
\item \textbf{Latency}: 0.03--0.05 ms using QPC microsecond counters.
\end{itemize}
\end{tcolorbox}

\end{multicols}

\vspace{-3mm}

% Key Architectural Conclusions
\begin{tcolorbox}[colback=accentgray,colframe=gray!50,arc=2mm,boxrule=0.5pt,top=1mm,bottom=1mm]
\scriptsize
\textbf{Key Takeaways \& Tradeoffs}:
1. \textbf{Complexity Invariance}: Cyclomatic complexity (CCN) reflects algorithmic branching (e.g., Composite condition evaluation = 8), remaining largely invariant across languages regardless of syntax.
2. \textbf{Developer Velocity vs Execution Control}: Python and JavaScript drastically reduce boilerplate for Observer and Strategy, while C++ delivers deterministic memory management and zero runtime overhead for embedded IoT edge nodes.
3. \textbf{Test Fidelity}: 100\% test pass rate achieved across all 24 runners (6/6 on Composite, Memento, Strategy; 5/5 on Observer, Factory, Template) verifying rigorous functional parity.
\end{tcolorbox}
