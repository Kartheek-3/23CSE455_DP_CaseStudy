import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

class NotFound extends RuntimeException {
    NotFound(String m) { super(m); }
}

class Account {                                   // TransferObject
    final long id;
    final double balance;
    Account(long id, double balance) { this.id = id; this.balance = balance; }
}

interface AccountDao {                            // DataAccessObject (abstraction: find + list only, as in DAO<T>)
    Account findById(long id);
    List<Account> getAll();
}

class InMemoryAccountDao implements AccountDao {  // ConcreteDAO (save exists only on the concrete class)
    private final Map<Long, Account> rows = new LinkedHashMap<>();
    private long next = 1;

    public Account findById(long id) {
        Account a = rows.get(id);
        if (a == null) throw new NotFound(String.valueOf(id));
        return a;
    }

    long save(double balance) {
        Account a = new Account(next++, balance);
        rows.put(a.id, a);
        return a.id;
    }

    public List<Account> getAll() { return new ArrayList<>(rows.values()); }
}

class AccountService {                            // BusinessObject: never touches storage directly
    private final InMemoryAccountDao dao;
    AccountService(InMemoryAccountDao dao) { this.dao = dao; }
    long open(double balance) { return dao.save(balance); }
    double balanceOf(long id) { return dao.findById(id).balance; }
    List<Double> balances() {
        List<Double> out = new ArrayList<>();
        for (Account a : dao.getAll()) out.add(a.balance);
        return out;
    }
}

class AccountDao {                                 // DataAccessObject (abstraction: find + list only, as in DAO<T>)
public:
    virtual ~AccountDao() = default;
    virtual const Account& findById(long id) const = 0;
    virtual std::vector<Account> getAll() const = 0;
};

class InMemoryAccountDao : public AccountDao {     // ConcreteDAO (save exists only on the concrete class)
    std::map<long, Account> rows_;
    long next_ = 1;
public:
    const Account& findById(long id) const override {
        auto it = rows_.find(id);
        if (it == rows_.end()) throw NotFound();
        return it->second;
    }
    long save(double balance) {
        rows_[next_] = Account{next_, balance};
        return next_++;
    }
    std::vector<Account> getAll() const override {
        std::vector<Account> out;
        for (const auto& kv : rows_) out.push_back(kv.second);
        return out;
    }
};

public:

    double balanceOf(long id) const { return dao_.findById(id).balance; }
    std::vector<double> balances() const {
        std::vector<double> out;

import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

class NoUser extends RuntimeException { }
class NotFound extends RuntimeException { }

class UserRepo {                               // Collaborator
    private final Set<Long> ids = new HashSet<>();
    void add(long id) { ids.add(id); }
    void require(long id) { if (!ids.contains(id)) throw new NoUser(); }
}

class AccountRepo {                            // Collaborator
    private final List<Double> balances = new ArrayList<>();
    long save(long userId, double balance) { balances.add(balance); return balances.size(); }
    double get(long id) {
        if (id < 1 || id > balances.size()) throw new NotFound();
        return balances.get((int) id - 1);
    }
}

class BankService {                            // ServiceLayer: one boundary that coordinates the use cases
    private final UserRepo users;
    private final AccountRepo accounts;
    BankService(UserRepo users, AccountRepo accounts) { this.users = users; this.accounts = accounts; }
    void createUser(long id) { users.add(id); }
    long openAccount(long userId, double balance) {
        users.require(userId);                 // use case: look up the owner first,
        return accounts.save(userId, balance); // then store the account
    }
    double balance(long accountId) { return accounts.get(accountId); }
}

import java.util.Map;

class BadRequest extends RuntimeException { }
class NotFound extends RuntimeException { }

class Row {                                     // RowData: data only, no behaviour (anemic)
    final String kind;
    double balance;
    Row(String kind, double balance) { this.kind = kind; this.balance = balance; }
}

class AccountScripts {                          // host of the TransactionScripts: one procedure per use case
    private final Map<Long, Row> rows = new HashMap<>();
    private long next = 1;

    long open(String kind, double balance) {
        rows.put(next, new Row(kind, balance));
        return next++;
    }

    private Row find(long id) {
        Row r = rows.get(id);
        if (r == null) throw new NotFound();
        return r;
    }

    double deposit(long id, double amount) {    // script: all rules are inline
        Row row = find(id);
        if (amount > 0) {
            double newBalance = row.balance + amount;
            if (row.kind.equals("S")) newBalance += 0.5;
            row.balance = newBalance;
            return newBalance;
        }
        throw new BadRequest();
    }

    double withdraw(long id, double amount) {   // script: all rules are inline
        Row row = find(id);
        if (amount > 0 && row.balance > amount) {
            row.balance -= amount;
            return row.balance;
        }
        throw new BadRequest();
    }
}
