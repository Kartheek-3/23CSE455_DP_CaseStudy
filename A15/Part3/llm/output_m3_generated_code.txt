import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Locale;
import java.util.Map;
import java.util.StringJoiner;

class BadRequest extends RuntimeException { }

class Event {                                    // Event (what is published)
    final String kind; final long accountId; final double amount;
    Event(String kind, long accountId, double amount) { this.kind = kind; this.accountId = accountId; this.amount = amount; }
}

interface Observer { void onEvent(Event e); }

class EventPublisher {                           // SubjectChannel: keeps the observers, knows none by type
    private final List<Observer> observers = new ArrayList<>();
    void subscribe(Observer o) { observers.add(o); }
    void publish(Event e) { for (Observer o : observers) o.onEvent(e); }
}

class AccountService {                           // Subject: publishes after a valid operation
    private final EventPublisher publisher;
    AccountService(EventPublisher publisher) { this.publisher = publisher; }
    void deposit(long accountId, double amount) { apply("DEPOSIT", accountId, amount); }
    void withdraw(long accountId, double amount) { apply("WITHDRAWAL", accountId, amount); }
    private void apply(String kind, long accountId, double amount) {
        if (amount <= 0) throw new BadRequest();  // failed operations publish nothing
        publisher.publish(new Event(kind, accountId, amount));
    }
}

class AuditLog implements Observer {             // ConcreteObserver
    private final Map<Long, List<Event>> entries = new HashMap<>();
    public void onEvent(Event e) { entries.computeIfAbsent(e.accountId, k -> new ArrayList<>()).add(e); }
    String text(long accountId) {
        StringJoiner j = new StringJoiner(",");
        for (Event e : entries.getOrDefault(accountId, List.of())) j.add(String.format(Locale.ROOT, "%s:%.2f", e.kind, e.amount));
        return j.length() == 0 ? "-" : j.toString();
    }
}

class HighValueAlert implements Observer {       // ConcreteObserver
    static final double THRESHOLD = 10000.0;
    private final List<Event> alerts = new ArrayList<>();
    public void onEvent(Event e) { if (e.amount >= THRESHOLD) alerts.add(e); }
    String text() {
        StringJoiner j = new StringJoiner(",");
        for (Event e : alerts) j.add(String.format(Locale.ROOT, "%s:%.2f@%d", e.kind, e.amount, e.accountId));
        return j.length() == 0 ? "-" : j.toString();
    }
}

#include <cstdio>
#include <map>
#include <stdexcept>

struct BadRequest : std::runtime_error { BadRequest() : std::runtime_error("bad request") {} };

struct Event {                                   // Event (what is published)
    std::string kind; long accountId; double amount;
};

struct Observer {
    virtual ~Observer() = default;
    virtual void onEvent(const Event& e) = 0;
};

class EventPublisher {                           // SubjectChannel: keeps the observers, knows none by type
    std::vector<Observer*> observers_;
public:
    void subscribe(Observer* o) { observers_.push_back(o); }
    void publish(const Event& e) { for (auto* o : observers_) o->onEvent(e); }
};

class AccountService {                           // Subject: publishes after a valid operation
    EventPublisher& publisher_;
    void apply(const std::string& kind, long id, double amount) {
        if (amount <= 0) throw BadRequest();     // failed operations publish nothing
        publisher_.publish(Event{kind, id, amount});
    }
public:
    explicit AccountService(EventPublisher& p) : publisher_(p) {}
    void deposit(long id, double amount) { apply("DEPOSIT", id, amount); }
    void withdraw(long id, double amount) { apply("WITHDRAWAL", id, amount); }
};

inline std::string money(double v) { char b[64]; std::snprintf(b, sizeof b, "%.2f", v); return b; }

class AuditLog : public Observer {               // ConcreteObserver
    std::map<long, std::vector<Event>> entries_;
public:
    void onEvent(const Event& e) override { entries_[e.accountId].push_back(e); }
    std::string text(long id) const {
        std::string out;
        auto it = entries_.find(id);
        if (it != entries_.end())
            for (const auto& e : it->second) out += (out.empty() ? "" : ",") + e.kind + ":" + money(e.amount);
        return out.empty() ? "-" : out;
    }
};

public:

import java.util.Map;

class BadRequest extends RuntimeException { }
class NotFound extends RuntimeException { }

class Account {                                  // DomainObject: data AND the rules that protect it
    static final double OVERDRAFT_LIMIT = 500.0;
    private final String kind;
    private double balance;
    Account(String kind, double balance) { this.kind = kind; this.balance = balance; }
    double balance() { return balance; }

    void deposit(double amount) {
        if (amount <= 0) throw new BadRequest();
        balance += amount + (kind.equals("S") ? 0.5 : 0.0);
    }

    void withdraw(double amount) {
        boolean allowed = kind.equals("C") ? balance - amount >= -OVERDRAFT_LIMIT : amount <= balance;
        if (amount <= 0 || !allowed) throw new BadRequest();
        balance -= amount;
    }

    void credit(double amount) {
        if (amount <= 0) throw new BadRequest();
        balance += amount;
    }
}

class Bank {                                     // Orchestrator: loads, calls the domain object, nothing more
    private final Map<Long, Account> accounts = new HashMap<>();
    private long next = 1;

    long open(String kind, double balance) { accounts.put(next, new Account(kind, balance)); return next++; }

    private Account find(long id) {
        Account a = accounts.get(id);
        if (a == null) throw new NotFound();
        return a;
    }

    double deposit(long id, double amount) { Account a = find(id); a.deposit(amount); return a.balance(); }
    double withdraw(long id, double amount) { Account a = find(id); a.withdraw(amount); return a.balance(); }

    void transfer(long fromId, long toId, double amount) {
        if (fromId == toId || amount <= 0) throw new BadRequest();
        Account source = find(fromId), target = find(toId);
        source.withdraw(amount);                 // throws before anything is changed
        target.credit(amount);
    }

    double balance(long id) { return find(id).balance(); }
}
