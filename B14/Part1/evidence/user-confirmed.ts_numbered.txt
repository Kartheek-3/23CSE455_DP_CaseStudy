   1: import { Resend } from 'resend'
   2: 
   3: export interface NotificationProfile {
   4:   id: string
   5:   email: string
   6:   full_name: string | null
   7:   confirmed_at: string | null
   8: }
   9: 
  10: function escapeHtml(str: string): string {
  11:   return str
  12:     .replace(/&/g, '&amp;')
  13:     .replace(/</g, '&lt;')
  14:     .replace(/>/g, '&gt;')
  15:     .replace(/"/g, '&quot;')
  16: }
  17: 
  18: function sanitizeSubject(str: string): string {
  19:   // Strip CR, LF, and other control characters to prevent header injection
  20:   return str.replace(/[\r\n\x00-\x1F\x7F]/g, '')
  21: }

  23: export async function sendAdminEmail(profile: NotificationProfile): Promise<void> {
  24:   const apiKey = process.env.RESEND_API_KEY
  25:   const fromEmail = process.env.RESEND_FROM_EMAIL
  26:   const adminEmail = process.env.ADMIN_EMAIL
  27:   if (!apiKey) throw new Error('RESEND_API_KEY is not configured')
  28:   if (!fromEmail) throw new Error('RESEND_FROM_EMAIL is not configured')
  29:   if (!adminEmail) throw new Error('ADMIN_EMAIL is not configured')
  30: 
  31:   const resend = new Resend(apiKey)
  32: 
  33:   const displayName = escapeHtml(profile.full_name ?? profile.email)
  34:   const email = escapeHtml(profile.email)
  35:   const id = escapeHtml(profile.id)
  36:   const confirmedAt = escapeHtml(profile.confirmed_at ?? '')
  37: 
  38:   await resend.emails.send({
  39:     from: fromEmail,
  40:     to: adminEmail,
  41:     subject: `New user registered: ${sanitizeSubject(profile.full_name ?? profile.email)}`,
  42:     html: `
  43:       <h2>New user registered</h2>
  44:       <p><strong>Name:</strong> ${displayName}</p>
  45:       <p><strong>Email:</strong> ${email}</p>
  46:       <p><strong>User ID:</strong> ${id}</p>
  47:       <p><strong>Confirmed at:</strong> ${confirmedAt}</p>
  48:     `,
  49:   })
  50: }

  52: export async function sendSlackNotification(profile: NotificationProfile): Promise<void> {
  53:   const webhookUrl = process.env.SLACK_WEBHOOK_URL
  54:   if (!webhookUrl) throw new Error('SLACK_WEBHOOK_URL is not configured')
  55: 
  56:   const displayName = profile.full_name ?? profile.email
  57:   const message = `New user confirmed: ${displayName} (${profile.email}) — ID: ${profile.id} — Confirmed at: ${profile.confirmed_at ?? 'N/A'}`
  58: 
  59:   const response = await fetch(webhookUrl, {
  60:     method: 'POST',
  61:     headers: { 'Content-Type': 'application/json' },
  62:     body: JSON.stringify({
  63:       text: 'New user confirmed',
  64:       blocks: [
  65:         {
  66:           type: 'section',
  67:           text: { type: 'plain_text', text: message },
  68:         },
  69:       ],
  70:     }),
  71:   })
  72: 
  73:   if (!response.ok) {
  74:     throw new Error(`Slack webhook failed: ${response.status} ${response.statusText}`)
  75:   }
  76: }
