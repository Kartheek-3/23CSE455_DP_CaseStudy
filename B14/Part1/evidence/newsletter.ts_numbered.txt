   1: import { Resend } from 'resend'
   2: import type { NewsletterSubscription } from '@/features/newsletter/types'
   3: 
   4: function escapeHtml(str: string): string {
   5:   return str
   6:     .replace(/&/g, '&amp;')
   7:     .replace(/</g, '&lt;')
   8:     .replace(/>/g, '&gt;')
   9:     .replace(/"/g, '&quot;')
  10: }
  11: 
  12: function sanitizeSubject(str: string): string {
  13:   // Strip CR, LF, and other control characters to prevent header injection
  14:   return str.replace(/[\r\n\x00-\x1F\x7F]/g, '')
  15: }
  16: 
  17: export interface PostEmailData {
  18:   title: string
  19:   slug: string
  20:   excerpt: string | null
  21:   cover_image: string | null
  22:   cover_image_alt?: string | null
  23: }
  24: 
  25: export async function sendNewsletterEmail(
  26:   subscriber: NewsletterSubscription,
  27:   post: PostEmailData
  28: ): Promise<void> {
  29:   const apiKey = process.env.RESEND_API_KEY
  30:   const fromEmail = process.env.RESEND_FROM_EMAIL
  31:   if (!apiKey) throw new Error('RESEND_API_KEY is not configured')
  32:   if (!fromEmail) throw new Error('RESEND_FROM_EMAIL is not configured')
  33: 
  34:   const siteUrl = (process.env.NEXT_PUBLIC_SITE_URL ?? 'http://localhost:3000').replace(/\/+$/, '')
  35:   const resend = new Resend(apiKey)
  36: 
  37:   const postUrl = `${siteUrl}/blog/${post.slug}`
  38:   const unsubscribeUrl = `${siteUrl}/api/newsletter/unsubscribe?token=${subscriber.unsubscribe_token}`
  39:   const title = escapeHtml(post.title)
  40:   const excerpt = post.excerpt ? escapeHtml(post.excerpt) : ''
  41: 
  42:   const { error } = await resend.emails.send({
  43:     from: fromEmail,
  44:     to: subscriber.email,
  45:     subject: sanitizeSubject(`New post: ${post.title}`),
  46:     html: `
  47:       <div style="font-family:sans-serif;max-width:600px;margin:0 auto;padding:24px;">
  48:         <h1 style="font-size:24px;font-weight:700;margin-bottom:8px;">${title}</h1>
  49:         ${excerpt ? `<p style="color:#6b7280;margin-bottom:16px;">${excerpt}</p>` : ''}
  50:         ${post.cover_image ? `<img src="${escapeHtml(post.cover_image)}" alt="${escapeHtml(post.cover_image_alt || post.title)}" style="width:100%;border-radius:8px;margin-bottom:16px;" />` : ''}
  51:         <a href="${escapeHtml(postUrl)}" style="display:inline-block;background:#111;color:#fff;padding:12px 24px;border-radius:6px;text-decoration:none;font-weight:600;">Read Post</a>
  52:         <hr style="margin:32px 0;border:none;border-top:1px solid #e5e7eb;" />
  53:         <p style="color:#9ca3af;font-size:12px;">
  54:           You're receiving this because you subscribed to new posts.
  55:           <a href="${escapeHtml(unsubscribeUrl)}" style="color:#9ca3af;">Unsubscribe</a>
  56:         </p>
  57:       </div>
  58:     `,
  59:   })
  60:   if (error) throw new Error(`Newsletter delivery failed: ${error.message}`)
  61: }
